cmake_minimum_required(VERSION 3.20)
project(MoveToMonitorButton VERSION 0.1.5 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(CTest)
option(MTMB_SANITIZE "Enable sanitizers for Linux API simulations" OFF)
set(MTMB_CORE src/app.cpp src/placement.cpp src/diagnostics.cpp)

function(strict_warnings target)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /utf-8 /permissive-)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
    endif()
endfunction()

if(WIN32)
    if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "MSVC")
        message(FATAL_ERROR "Windows distribution builds require MSVC and the official Windows SDK")
    endif()
    if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_GENERATOR_PLATFORM STREQUAL "x64")
        message(FATAL_ERROR "Use the Visual Studio generator with -A x64")
    endif()
    enable_language(RC)
    configure_file(assets/app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    add_executable(MoveToMonitorButton WIN32 src/main.cpp ${MTMB_CORE} "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
    target_include_directories(MoveToMonitorButton PRIVATE src)
    target_compile_definitions(MoveToMonitorButton PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
    target_link_libraries(MoveToMonitorButton PRIVATE user32 gdi32 shell32 dwmapi advapi32)
    strict_warnings(MoveToMonitorButton)
    target_compile_options(MoveToMonitorButton PRIVATE /guard:cf)
    target_link_options(MoveToMonitorButton PRIVATE /MANIFEST:NO /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /guard:cf)
    file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/$<CONFIG>/build-info.json" CONTENT
"{\n  \"version\": \"${PROJECT_VERSION}\",\n  \"compiler\": \"${CMAKE_CXX_COMPILER_ID}\",\n  \"compilerVersion\": \"${CMAKE_CXX_COMPILER_VERSION}\",\n  \"windowsSdk\": \"${CMAKE_VS_WINDOWS_TARGET_PLATFORM_VERSION}\",\n  \"generator\": \"${CMAKE_GENERATOR}\",\n  \"configuration\": \"$<CONFIG>\",\n  \"architecture\": \"x64\",\n  \"runtime\": \"MultiThreaded$<$<CONFIG:Debug>:Debug>\",\n  \"officialSdk\": true,\n  \"customAbi\": false\n}\n")
endif()

if(BUILD_TESTING)
    add_executable(layout_test tests/layout_test.cpp)
    target_include_directories(layout_test PRIVATE src)
    strict_warnings(layout_test)
    add_test(NAME layout COMMAND layout_test)
    if(WIN32)
        add_executable(sdk_abi_test tests/abi_test.cpp)
        target_include_directories(sdk_abi_test PRIVATE src)
        strict_warnings(sdk_abi_test)
        add_test(NAME sdk_abi COMMAND sdk_abi_test)
        add_executable(sdk_helpers_test tests/sdk_helpers_test.cpp src/diagnostics.cpp)
        target_include_directories(sdk_helpers_test PRIVATE src)
        target_link_libraries(sdk_helpers_test PRIVATE user32 gdi32 shell32 advapi32)
        strict_warnings(sdk_helpers_test)
        add_test(NAME sdk_helpers COMMAND sdk_helpers_test)
        add_test(NAME executable_startup COMMAND MoveToMonitorButton --startup-check)
        set_tests_properties(executable_startup PROPERTIES TIMEOUT 15)
    elseif(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        # Same implementation, separate translation units. No inclusion of main.cpp.
        add_executable(window_logic_test tests/window_logic_test.cpp ${MTMB_CORE})
        target_include_directories(window_logic_test PRIVATE src tests)
        target_compile_definitions(window_logic_test PRIVATE MTMB_TEST_API)
        # Native host wchar_t is essential now that we use the standard string library.
        # These are API behavior doubles, NOT declarations used in a Windows binary.
        target_compile_options(window_logic_test PRIVATE -fms-extensions -Wno-ignored-attributes)
        strict_warnings(window_logic_test)
        add_test(NAME window_logic_simulation COMMAND window_logic_test)
    endif()
    if(MTMB_SANITIZE AND NOT WIN32)
        foreach(target layout_test window_logic_test)
            if(TARGET ${target})
                target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
                target_link_options(${target} PRIVATE -fsanitize=address,undefined)
            endif()
        endforeach()
    endif()
endif()
